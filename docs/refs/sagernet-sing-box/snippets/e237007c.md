The deprecated format used a top-level fakeip object, while the new format integrates fakeip settings directly into the server definition.

```json
{
  "dns": {
    "servers": [
      {
        "address": "1.1.1.1"
      },
      {
        "address": "fakeip",
        "tag": "fakeip"
      }
    ],
    "rules": [
      {
        "query_type": [
          "A",
          "AAAA"
        ],
        "server": "fakeip"
      }
    ],
    "fakeip": {
      "enabled": true,
      "inet4_range": "198.18.0.0/15",
      "inet6_range": "fc00::/18"
    }
  }
}
```

```json
{
  "dns": {
    "servers": [
      {
        "type": "udp",
        "server": "1.1.1.1"
      },
      {
        "type": "fakeip",
        "tag": "fakeip",
        "inet4_range": "198.18.0.0/15",
        "inet6_range": "fc00::/18"
      }
    ],
    "rules": [
      {
        "query_type": [
          "A",
          "AAAA"
        ],
        "server": "fakeip"
      }
    ]
  }
}
```
