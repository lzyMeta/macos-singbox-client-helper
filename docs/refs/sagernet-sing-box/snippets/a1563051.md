The legacy DNS server configuration is deprecated in favor of explicit server types and rule-based routing.

```json
{
  "dns": {
    "servers": [
      {
        "address": "1.1.1.1"
      },
      {
        "tag": "google",
        "address": "8.8.8.8",
        "client_subnet": "1.1.1.1"
      }
    ]
  }
}
```

```json
{
  "dns": {
    "servers": [
      {
        "type": "udp",
        "server": "1.1.1.1"
      },
      {
        "type": "udp",
        "tag": "google",
        "server": "8.8.8.8"
      }
    ],
    "rules": [
      {
        "domain": "google.com",
        "server": "google",
        "client_subnet": "1.1.1.1"
      }
    ]
  }
}
```
